# Cybersecurity Network Monitoring System - Requirements
# =======================================================

# No external third-party packages (pip dependencies) are required.

# - time        (Latency & response time measurement)
# - datetime    (Timestamping operations)
# - csv         (CSV security report export)
# - platform   (OS detection for Windows/Kali Linux compatibility)
# - re         (Regex parsing for ping output)
# - os / sys   (File system & process management)

python_version >= 3.6
